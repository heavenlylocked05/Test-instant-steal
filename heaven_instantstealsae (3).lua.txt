local function askDoffWithRetry(waitForDone)
    if not AskDoff then return true end
    local MAX_ATTEMPTS, RETRY_DELAY = 3, 0.5
    local function run()
        local success = false
        for attempt = 1, MAX_ATTEMPTS do
            local ok, result = pcall(function() return AskDoff:InvokeServer() end)
            if ok and result then success = true; break end
            if attempt < MAX_ATTEMPTS then task.wait(RETRY_DELAY) end
        end
        return success
    end
    if waitForDone then return run() else task.spawn(run); return true end
end

UserInputService.JumpRequest:Connect(function() askDoffWithRetry(false) end)
UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    if input.KeyCode == Enum.KeyCode.Space then askDoffWithRetry(false) end
end)

local function loadPersistData()
    pcall(function()
        if type(isfile) ~= "function" or type(readfile) ~= "function" then return end
        if not isfile(CONFIG.SAVE_FILE) then return end
        local content = readfile(CONFIG.SAVE_FILE)
        if not content or content == "" then return end
        local data = HttpService:JSONDecode(content)
        if type(data) ~= "table" then return end

        if type(data.history) == "table" then
            table.clear(S.history)
            for _, h in ipairs(data.history) do
                if type(h) == "table" and h.Name then
                    local col = Color3.fromRGB(170, 190, 200)
                    local c = h.RarityColor
                    if type(c) == "table" and #c >= 3 then
                        col = Color3.new(c[1], c[2], c[3])
                    end
                    table.insert(S.history, {
                        Name = tostring(h.Name or ""),
                        Rarity = tostring(h.Rarity or ""),
                        RarityColor = col,
                        Generation = tonumber(h.Generation) or 0,
                        Time = tostring(h.Time or ""),
                    })
                end
            end
            while #S.history > CONFIG.HISTORY_MAX do table.remove(S.history) end
        end

        if data.stealMode == "admin" or data.stealMode == "instant" then
            S.stealMode = data.stealMode
        end

        if type(data.rarityFilter) == "table" then
            for k, v in pairs(data.rarityFilter) do
                if type(k) == "string" and type(v) == "boolean" then
                    S.rarityFilter[k] = v
                end
            end
        end
    end)
end

savePersistData = function()
    pcall(function()
        if type(writefile) ~= "function" then return end
        local hist = {}
        for i, h in ipairs(S.history) do
            local c = h.RarityColor or Color3.fromRGB(170, 190, 200)
            hist[i] = {
                Name = tostring(h.Name or ""),
                Rarity = tostring(h.Rarity or ""),
                RarityColor = { c.R, c.G, c.B },
                Generation = tonumber(h.Generation) or 0,
                Time = tostring(h.Time or ""),
            }
        end
        local filter = {}
        for k, v in pairs(S.rarityFilter) do
            if v then filter[k] = true end
        end
        writefile(CONFIG.SAVE_FILE, HttpService:JSONEncode({
            version = 1,
            history = hist,
            stealMode = S.stealMode or "instant",
            rarityFilter = filter,
        }))
    end)
end

local function dl()

    if not cy then return nil end
    local t = workspace:GetServerTimeNow()
    local o1, n = pcall(cy.IsNightPhase, t)
    local o2, ns = pcall(cy.NightStartTime, t)
    local o3, nr = pcall(cy.NextResetTime, t)

    local period = cy.ResetPeriodSeconds or 0
    if n or t - (nr - period) < 5 then return 0 end
    return ns - t
end

local hiddenChar = {}
local function hideCharacter()
    table.clear(hiddenChar)
    local ch = lp.Character
    if not ch then return end
    for _, obj in ch:GetDescendants() do
        if obj:IsA("BasePart") then
            hiddenChar[obj] = { "LocalTransparencyModifier", obj.LocalTransparencyModifier }
            pcall(function() obj.LocalTransparencyModifier = 1 end)
        elseif obj:IsA("Decal") then
            hiddenChar[obj] = { "Transparency", obj.Transparency }
            pcall(function() obj.Transparency = 1 end)
        end
    end
end

local function showCharacter()
    for obj, data in hiddenChar do
        if obj and obj.Parent then pcall(function() obj[data[1]] = data[2] end) end
    end
    table.clear(hiddenChar)
end

local trailKeywords = { "Trail", "Path", "Route", "Guide", "Marker", "Waypoint", "Footstep", "Line" }
local hiddenTrail = {}
local function isTrailVisual(obj)
    if not obj or not obj.Parent then return false end
    if not (obj:IsA("BasePart") or obj:IsA("Decal") or obj:IsA("Trail")
        or obj:IsA("Beam") or obj:IsA("ParticleEmitter") or obj:IsA("Highlight")) then return false end
    local n = obj.Name
    for _, k in trailKeywords do if n:find(k, 1, true) then return true end end
    return false
end

local function hideTrailObjects()
    table.clear(hiddenTrail)
    for _, obj in workspace:GetDescendants() do
        if isTrailVisual(obj) then
            if obj:IsA("BasePart") then
                if obj.LocalTransparencyModifier ~= 1 then
                    hiddenTrail[obj] = { "LocalTransparencyModifier", obj.LocalTransparencyModifier }
                    pcall(function() obj.LocalTransparencyModifier = 1 end)
                end
            elseif obj:IsA("Decal") then
                if obj.Transparency ~= 1 then
                    hiddenTrail[obj] = { "Transparency", obj.Transparency }
                    pcall(function() obj.Transparency = 1 end)
                end
            else
                if obj.Enabled then
                    hiddenTrail[obj] = { "Enabled", obj.Enabled }
                    pcall(function() obj.Enabled = false end)
                end
            end
        end
    end
end

local function showTrailObjects()
    for obj, data in hiddenTrail do
        if obj and obj.Parent then pcall(function() obj[data[1]] = data[2] end) end
    end
    table.clear(hiddenTrail)
end

local visualClone = nil
local function removeVisualClone()
    if visualClone then pcall(function() visualClone:Destroy() end); visualClone = nil end
end

local function createVisualCloneAt(position, lookAtPos)
    removeVisualClone()
    local ch = lp.Character
    if not ch or not position then return end

    local wasArchivable = ch.Archivable
    if not wasArchivable then ch.Archivable = true end
    local ok, clone = pcall(function() return ch:Clone() end)
    ch.Archivable = wasArchivable
    if not ok or not clone then return end

    visualClone = clone
    for _, obj in ipairs(clone:GetDescendants()) do
        if obj:IsA("Script") or obj:IsA("LocalScript") or obj:IsA("ModuleScript") then
            pcall(function() obj:Destroy() end)
        end
    end

    local hr = clone:FindFirstChild("HumanoidRootPart")
    if hr and hr:IsA("BasePart") then
        if lookAtPos then
            local dir = Vector3.new(lookAtPos.X - position.X, 0, lookAtPos.Z - position.Z)
            if dir.Magnitude > 0.1 then
                hr.CFrame = CFrame.lookAt(position, position + dir.Unit)
            else
                hr.CFrame = CFrame.new(position)
            end
        else
            hr.CFrame = CFrame.new(position)
        end
    end

    for _, obj in ipairs(clone:GetDescendants()) do
        if obj:IsA("BasePart") then
            pcall(function()
                obj.Anchored = true
                obj.CanCollide = false
                obj.CanQuery = false
                obj.CanTouch = false
            end)
        end
    end

    local hum = clone:FindFirstChildOfClass("Humanoid")
    if hum then
        pcall(function()
            hum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None
            hum.NameDisplayDistance = 0
            hum.HealthDisplayDistance = 0
            hum.WalkSpeed = 0
            hum.JumpPower = 0
            hum.AutoRotate = false
            hum:ChangeState(Enum.HumanoidStateType.Physics)
        end)
    end

    clone.Name = "ZHVisualClone"
    clone.Parent = workspace
end

local cameraLockCF = nil
local cameraLockConn = nil

local function lockCameraAt(target, from)
    local cam = workspace.CurrentCamera
    if not cam then return end
    local origin = from or cam.CFrame.Position
    local dir = Vector3.new(target.X - origin.X, 0, target.Z - origin.Z)
    if dir.Magnitude < 1 then dir = Vector3.new(0, 0, 1) end
    dir = dir.Unit
    local offset = -dir * CONFIG.CAM_DIST + Vector3.new(0, CONFIG.CAM_HEIGHT, 0)
    cameraLockCF = CFrame.lookAt(target + offset, target)
    cam.CameraType = Enum.CameraType.Scriptable
    cam.CFrame = cameraLockCF
    createVisualCloneAt(cameraLockCF.Position, target)
    if cameraLockConn then return end
    cameraLockConn = RunService.RenderStepped:Connect(function()
        if not cameraLockCF then return end
        local c = workspace.CurrentCamera
        if c then c.CameraType = Enum.CameraType.Scriptable; c.CFrame = cameraLockCF end
    end)
end

local function unlockCamera()
    cameraLockCF = nil
    if cameraLockConn then cameraLockConn:Disconnect(); cameraLockConn = nil end
    local cam = workspace.CurrentCamera
    if not cam then return end
    cam.CameraType = Enum.CameraType.Custom
    local _, hu = rt()
    if hu then cam.CameraSubject = hu end
end

local function sw()
    local ch = lp.Character
    local hu = ch and ch:FindFirstChildOfClass("Humanoid")
    if st.sc and hu == st.sc and st.sc.Parent == ch then return true end
    if st.sc then
        if st.so then pcall(function() if st.so.Parent == nil then st.so.Parent = ch end end) end
        if st.sc then pcall(function() st.sc:Destroy() end) end
        st.so, st.sc = nil, nil
        ul()
    end
    if not hu or hu.Health <= 0 or hu.FloorMaterial == Enum.Material.Air or not st.gd[hu:GetState()] then
        return false
    end
    local c = hu:Clone()
    hu.Parent = nil
    c.Parent = ch
    workspace.CurrentCamera.CameraSubject = c
    ct(c)
    ra(ch)
    st.so, st.sc, st.at = hu, c, nil
    local a, b = hu:FindFirstChildOfClass("Animator"), c:FindFirstChildOfClass("Animator")
    if a and b then
        st.sl[#st.sl + 1] = a.AnimationPlayed:Connect(function(k)
            if not k.Animation or c.Parent == nil then return end
            local ok, t = pcall(function() return b:LoadAnimation(k.Animation) end)
            if not ok or not t then return end
            pcall(function()
                t.Priority, t.Looped = k.Priority, k.Looped
                t:Play(0.05, math.max(k.WeightTarget, 0.01), k.Speed)
            end)
            local s
            s = k.Stopped:Connect(function()
                s:Disconnect()
                pcall(function() t:Stop(0.1) end)
            end)
        end)
    end
    st.sl[#st.sl + 1] = c.Died:Connect(function()
        ul()
        st.so, st.sc = nil, nil
        local cc = lp.Character
        if cc and hu.Parent == nil then
            hu.Parent = cc
            workspace.CurrentCamera.CameraSubject = hu
            ct(hu)
        end
        pcall(function() c:Destroy() end)
        hu.Health = 0
    end)
    return true
end

local function mv(tg, ck, to, cp)
    local c = lp.Character
    if not rt() then return false, "Move Failed: No Character" end
    local t0, dn, wy = os.clock(), nil, nil
    local kp, kt = rt().Position, os.clock()
    local cn = RunService.PreSimulation:Connect(function(d)
        if dn ~= nil then return end
        local hr, hm = rt()
        if isCancel() then dn, wy = false, "Stopped" return end
        if not hr or not hm or hm.Health <= 0 or lp.Character ~= c then
            dn, wy = false, "Move Failed: Character Lost" return
        end
        fz(hm.FloorMaterial == Enum.Material.Air)
        local r = ck and ck()
        if r then dn, wy = false, r return end
        if os.clock() - t0 > to then dn, wy = false, "Move Failed: Timeout" return end
        local v = tg - hr.Position
        local m = v.Magnitude
        if m <= 1 then hr.AssemblyLinearVelocity = Vector3.zero; dn = true; return end
        if os.clock() - kt >= 1 then
            if (hr.Position - kp).Magnitude < 2 then dn, wy = false, "Move Failed: Blocked" return end
            kp, kt = hr.Position, os.clock()
        end
        hr.AssemblyLinearVelocity = v.Unit * math.min(hm.WalkSpeed * (cp or 1), m / math.max(d, 1/240))
            + Vector3.new(0, workspace.Gravity * d * 0.5, 0)
        hr.AssemblyAngularVelocity = Vector3.zero
    end)
    st.cn[#st.cn + 1] = cn
    while dn == nil do RunService.Heartbeat:Wait() end
    pcall(function() cn:Disconnect() end)
    return dn, wy
end

local function hp(en, y, rq, ck)
    while true do
        if isCancel() then return false, "Stopped" end
        local r = ck and ck()
        if r then return false, r end
        local hr = rt()
        if not hr then return false, "Home Failed: Character Lost" end
        if rq and not cr() then return false, "Home Failed: Egg Lost" end
        fz(true)
        local d = Vector3.new(en.X - hr.Position.X, 0, en.Z - hr.Position.Z)
        if d.Magnitude <= 264 then
            hr.CFrame = CFrame.new(en) * hr.CFrame.Rotation
            hr.AssemblyLinearVelocity = Vector3.zero
            return true
        end
        hr.CFrame = CFrame.new(hr.Position.X + d.Unit.X * 264, y, hr.Position.Z + d.Unit.Z * 264) * hr.CFrame.Rotation
        hr.AssemblyLinearVelocity = Vector3.zero
        task.wait(0.1)
    end
end

local function go(e)
    local z0, z1 = cz()
    if not z0 then return false, "Fly Failed: No Corridor" end
    local h = rt()
    if not h then return false, "Fly Failed: No Character" end
    local hz = math.clamp(h.Position.Z, z0, z1)
    if hz ~= h.Position.Z then
        local ok, wy = mv(Vector3.new(h.Position.X, h.Position.Y, hz), nil, 10)
        if not ok then return false, wy end
        h = rt()
        if not h then return false, "Fly Failed: No Character" end
    end
    local a = Vector3.new(h.Position.X, 0, h.Position.Z)
    local b = Vector3.new(e.p.X, 0, math.clamp(e.p.Z, z0, z1))
    local y = math.max(h.Position.Y, e.p.Y) + 3
    local dh = (b - a).Magnitude
    local u = dh > 0.1 and (b - a).Unit or Vector3.zero
    local nt = 0
    local function ck()
        if os.clock() < nt then return nil end
        nt = os.clock() + 0.25
        if not av(e.u) then return "Fly Failed: Egg Gone" end
        local dd = dl()
        if not dd or dd < 2 then return "Fly Failed: Night Coming" end
        return nil
    end
    local STEP = CONFIG.FLY_STEP or 700
    local w = {}
    if dh > STEP then
        local segs = math.ceil(dh / STEP)
        for i = 1, segs - 1 do
            local p = a + u * math.min(STEP * i, dh)
            w[#w + 1] = Vector3.new(p.X, y, p.Z)
        end
    end
    for _, p in ipairs(w) do
        local hr, hu = rt()
        local q = Vector3.new(p.X, y, p.Z)
        local ok, wy = mv(q, ck, hr and hu and (q - hr.Position).Magnitude / math.max(hu.WalkSpeed * st.fm, 16) + 10 or 30, st.fm)
        if not ok then return false, wy end
    end
    return mv(e.p + Vector3.new(0, 3, 0), ck, 10, st.fm)
end

local function tk(e)
    local k = nil
    if si and type(si.LooksLikeFirstAreaUid) == "function" then
        local ok, first = pcall(si.LooksLikeFirstAreaUid, e.u)
        if ok and first and type(si.SlotKey) == "function" then
            local o2, sk = pcall(si.SlotKey, e.r.AreaId, e.r.NestId)
            if o2 then k = sk end
        end
    end
    local t0 = os.clock()
    while kd() and os.clock() - t0 < 5 do task.wait(0.1) end
    local t1 = os.clock()
    local ok, a, b = pcall(es.CarryFieldEgg, e.u, k)
    while ok and not a and (
        (tostring(b):find("knocked down", 1, true) and os.clock() - t0 < 6) or
        (tostring(b):find("Get closer", 1, true) and os.clock() - t1 < 1.5)
    ) do
        task.wait(0.25)
        ok, a, b = pcall(es.CarryFieldEgg, e.u, k)
    end
    if ok and not a and tostring(b):find("gameplay area", 1, true) then
        local l = sp()
        local o2, r = pcall(es.ReadFieldEgg, e.u)
        local p = o2 and type(r) == "table" and typeof(r.BoundsCFrame) == "CFrame" and r.BoundsCFrame.Position
        if l and p then
            local lv = l.CFrame.LookVector
            if mv(p - lv * (lv:Dot(p - l.Position) - 4) + Vector3.new(0, 3, 0), nil, 5) then
                ok, a, b = pcall(es.CarryFieldEgg, e.u, k)
            end
        end
    end
    if not ok then return false, "Take Failed: " .. tostring(a) end
    if not a then return false, "Take Failed: " .. tostring(b or "Denied") end
    local t = os.clock()
    repeat task.wait(0.05) until cr() or os.clock() - t > 1
    if not cr() then return false, "Take Failed: Not Carrying" end
    st.pu = e.u
    return true
end

local function rp()
    local t0 = os.clock()
    task.wait(st.dw)
    local o1, cs = pcall(es.ReadCarryState)
    local u = o1 and type(cs) == "table" and cs.Uid
    if type(u) ~= "string" then return false, "Drop Failed: No Carry Uid" end
    local o2, a, b = pcall(es.DropFieldEgg, "PlayerRequest")
    if not o2 or not a then return false, "Drop Failed: " .. tostring(o2 and (b or "Denied") or a) end
    local t1 = os.clock()
    repeat task.wait(0.03) until not cr() or os.clock() - t1 > 1
    if cr() then return false, "Drop Failed: Still Carrying" end
    local o3, r = pcall(es.ReadFieldEgg, u)
    local ok, wy = tk({ u = u, r = o3 and type(r) == "table" and r or {} })
    if not ok then return false, "Repick " .. tostring(wy) end
    return true
end

local function ze()
    local w = workspace:FindFirstChild("World") or workspace:FindFirstChild("__OBJECTS")
    local a = w and w:FindFirstChild("Areas")
    local g = a and a:FindFirstChild("GuardAreas")
    local o = {}
    if not g then return o end
    for _, z in g:GetChildren() do
        local b = z:FindFirstChild("Bounds")
        if b and b:IsA("BasePart") then o[#o + 1] = b.Position.X - b.Size.X / 2 end
    end
    table.sort(o, function(x1, x2) return x1 > x2 end)
    return o
end

local function sz(z, lq, lv)
    local tf = workspace:FindFirstChild("Transient")
    local hb = {}
    for _, d in tf and tf:GetDescendants() or {} do
        if d.Name == "PlayerTrap" then
            local x = d:FindFirstChild("Hitbox") or d
            if x:IsA("BasePart") and d:GetAttribute("Owner") ~= lp.Name then
                local s = lv:Dot(x.Position - lq)
                if s > -40 and s < 70 then
                    hb[#hb + 1] = { x.Position.Z, math.max(x.Size.X, x.Size.Z) / 2 + 6 }
                end
            end
        end
    end
    if #hb == 0 then return z end
    local z0, z1 = cz()
    z0, z1 = z0 or z - 60, z1 or z + 60
    for k = 0, 60 do
        for _, c in { z + k * 3, z - k * 3 } do
            if c >= z0 and c <= z1 then
                local ok = true
                for _, t in hb do
                    if math.abs(c - t[1]) < t[2] then ok = false; break end
                end
                if ok then return c end
            end
        end
    end
    return z
end

local function hm(rq)
    local l, h = sp(), rt()
    if not l then return false, "Home Failed: No Separation Line" end
    if not h then return false, "Home Failed: No Character" end
    local lv, lq = l.CFrame.LookVector, l.Position
    local q = h.Position - lv * (lv:Dot(h.Position - lq) - 48)
    local en = Vector3.new(q.X, lq.Y + 3, sz(q.Z, lq, lv))
    local y = lq.Y + 45
    if rq then
        local k = 0
        for _, x in ze() do
            if x < h.Position.X then
                k += 1
                if k % st.rz == 0 and x + 10 > en.X + 150 then
                    local ok, wy = hp(Vector3.new(x + 10, en.Y, q.Z), y, true)
                    if not ok then return false, wy end
                    ok, wy = rp()
                    if not ok then return false, wy end
                end
            end
        end
    end
    local ok, wy = hp(en, y, rq)
    if not ok then return false, wy end
    if not rq then
        task.wait(0.1)
        return mv(en - lv * 70, nil, 5)
    end
    ok, wy = rp()
    if not ok then return false, wy end
    local o1, cs = pcall(es.ReadCarryState)
    local sm = o1 and type(cs) == "table" and tonumber(cs.SpeedMultiplier) or 0.9
    local n0 = st.n
    ok, wy = mv(en - lv * 70, function()
        if not cr() then return "Delivered" end
        return nil
    end, 5, sm)
    if wy ~= "Delivered" and not ok then return false, wy end
    local t = os.clock()
    repeat task.wait(0.05) until (not cr() and st.n > n0) or os.clock() - t > 3
    if st.n <= n0 then return false, "Home Failed: Not Claimed" end
    return true
end

local function formatNum(n)
    n = tonumber(n) or 0
    local function trim(x)
        return (string.format("%.2f", x):gsub("(%..-)0+$", "%1"):gsub("%.$", ""))
    end
    local abs = math.abs(n)
    for _, p in ipairs({ {1e12,"T"}, {1e9,"B"}, {1e6,"M"}, {1e3,"K"} }) do
        if abs >= p[1] then return trim(n / p[1]) .. p[2] end
    end
    return tostring(math.floor(n + 0.5))
end

local rarityColorCache = {}
local function getRarityColor(name)
    local key = string.lower(tostring(name or "")):gsub("%s+", "")
    if rarityColorCache[key] then return rarityColorCache[key] end
    local color = Color3.fromRGB(170, 190, 200)
    if Rarity and Rarity.Rarities then
        for k, r in pairs(Rarity.Rarities) do
            local k1 = string.lower(tostring(k)):gsub("%s+", "")
            local k2 = string.lower(tostring(type(r) == "table" and r.DisplayName or "")):gsub("%s+", "")
            if k1 == key or k2 == key then
                local c = type(r) == "table" and r.Color
                if typeof(c) == "ColorSequence" then c = c.Keypoints[1].Value end
                if typeof(c) == "Color3" then color = c end
                break
            end
        end
    end
    rarityColorCache[key] = color
    return color
end

local function getScaleMultiplier(scale)
    local s = tonumber(scale) or 1
    if s <= 5 then return s ^ 1.85 end
    return (s / 5) ^ 1.2 * 19.637875755794113
end

local function buildEgg(record)
    local asset = dr[record.AssetCategory]
    if not asset then return nil end
    local baseRate = tonumber(asset.EarningRate) or 0
    local scale = tonumber(record.AssetScale) or 1
    local rarityName, color = "", Color3.fromRGB(170, 190, 200)
    if asset.Rarity then
        rarityName = asset.Rarity.DisplayName or asset.Rarity.Name or ""
        color = getRarityColor(rarityName)
    end
    return {
        Uid = record.Uid,
        Category = record.AssetCategory,
        Name = asset.DisplayName or record.AssetCategory,
        Generation = math.round(baseRate * getScaleMultiplier(scale)),
        RarityName = rarityName,
        RarityColor = color,
        Position = (record.BoundsCFrame and record.BoundsCFrame.Position) or record.Position,
        State = record.State,
        Raw = record,
    }
end

local lastStealCat = nil
local function updateStealCard()
    if not S.currentSteal or currentTab ~= "steal" then
        if stealCard.Visible then
            stealCard.Visible = false
            updatePanelLayout()
        end
        return
    end
    if not stealCard.Visible then
        stealCard.Visible = true
        updatePanelLayout()
    end
    local e = S.currentSteal.egg
    if e then
        stealName.Text = tostring(e.Name or "?")
        stealRarity.Text = tostring(e.RarityName or "")
        stealRarity.TextColor3 = e.RarityColor or Color3.fromRGB(170, 190, 200)
        stealMoney.Text = formatNum(e.Generation or 0) .. "/s"
        if lastStealCat ~= e.Category then
            lastStealCat = e.Category
            for _, c in ipairs(stealPreview:GetChildren()) do
                if not c:IsA("UICorner") then c:Destroy() end
            end
            task.spawn(function()
                task.wait(0.05)
                local AV = safeReq(RS, "Client", "AssetViewport")
                if not AV or not e.Category then return end
                local ok, vp = pcall(function() return AV.Mount(stealPreview) end)
                if not ok or not vp then return end
                vp.Size = UDim2.fromScale(1, 1)
                vp.Position = UDim2.fromScale(0.5, 0.5)
                vp.AnchorPoint = Vector2.new(0.5, 0.5)
                vp.BackgroundTransparency = 1
                vp.ZIndex = 16
                pcall(function() AV.ShowAsset(e.Category, vp, true) end)
            end)
        end
    end
    stealPhase.Text = tostring(S.currentSteal.phase or "Working")
    local pct = math.floor(S.currentSteal.percent or 0)
    stealPercent.Text = pct .. "%"
    TweenService:Create(stealBar, TweenInfo.new(0.15), { Size = UDim2.new(pct/100, 0, 1, 0) }):Play()
end

local function renderHistory()
    local n = #S.history
    historyTitle.Text = "STEAL HISTORY (" .. n .. ")"
    for i = 1, n do
        local item = historyPool[i]
        if not item then
            item = createHistoryItem()
            item.Frame.Parent = historyList
            item.Frame.LayoutOrder = i
            historyPool[i] = item
        end
        local e = S.history[i]
        item.Frame.Visible = true
        item.Name.Text = tostring(e.Name or "?")
        item.Rarity.Text = tostring(e.Rarity or "")
        item.Rarity.TextColor3 = e.RarityColor or Color3.fromRGB(170, 190, 200)
        item.Accent.BackgroundColor3 = e.RarityColor or SKY
        item.Gen.Text = formatNum(e.Generation or 0) .. "/s"
        item.Time.Text = tostring(e.Time or "")
    end
    for i = n + 1, #historyPool do

local function getTimeString()
    local ok, result = pcall(function()
        return os.date("%H:%M:%S") .. " - " .. os.date("%d/%m")
    end)
    if ok and result then return result end
    return tostring(os.time())
end

end

local function updateCard(c, egg)
    c.Frame.Visible = true
    c.Egg = egg
    if c.LastUid ~= egg.Uid or c.LastCat ~= egg.Category then
        c.LastUid = egg.Uid
        c.LastCat = egg.Category
        for _, child in ipairs(c.Preview:GetChildren()) do
            if not child:IsA("UICorner") then child:Destroy() end
        end
        attachViewport(c.Preview, egg.Category)
        c.NameLabel.Text = tostring(egg.Name or "UNKNOWN")
        c.RarityLabel.Text = tostring(egg.RarityName or "")
        c.RarityLabel.TextColor3 = egg.RarityColor or Color3.fromRGB(170, 190, 200)
    end
    c.EarnLabel.Text = formatNum(egg.Generation or 0)
    refreshCardVisual(c, egg)
    if S.selectedUid == egg.Uid then S.selectedEgg = egg end
end

local function renderCards()
    local total = math.min(#S.filtered, CONFIG.MAX_CARDS)
    for i = 1, total do updateCard(getCard(i), S.filtered[i]) end
    for i = total + 1, #cardPool do cardPool[i].Frame.Visible = false end
    emptyLabel.Visible = total == 0
    if total == 0 then emptyLabel.Text = Snapshot and "Scanning... no egg found" or "Snapshot remote NOT FOUND" end
end

local function refreshEggs(force)
    local now = tick()
    if not force and now - S.lastRefresh < CONFIG.REFRESH then return end
    if not Snapshot then return end
    local ok, result = pcall(function() return Snapshot:InvokeServer() end)
    if not ok then return end
    S.lastRefresh = now
    table.clear(S.eggs)
    if result and result.Records then
        for _, record in ipairs(result.Records) do
            if record.State == "Slot" or record.State == "Dropped" then
                local data = buildEgg(record)
                if data then S.eggs[#S.eggs + 1] = data end
            end
        end
    end
    table.sort(S.eggs, function(a, b) return a.Generation > b.Generation end)
    table.clear(S.filtered)
    for _, egg in ipairs(S.eggs) do
        if eggPass(egg) then S.filtered[#S.filtered + 1] = egg end
    end
    if S.selectedUid then
        local found = false
        for _, egg in ipairs(S.eggs) do
            if egg.Uid == S.selectedUid then S.selectedEgg = egg; found = true; break end
        end
        if not found then S.selectedUid = nil; S.selectedEgg = nil end
    end
end

-- PROGRESS MONITOR
local progressConn
local function startProgressMonitor()
    if progressConn then return end
    progressConn = RunService.Heartbeat:Connect(function()
        local cs = S.currentSteal
        if not cs then return end
        local hr = rt()
        if not hr then return end
        local ph = cs.phase
        local pct = cs.percent or 0
        if ph == "Flying" and cs.flyStart and cs.flyTarget then
            local total = (cs.flyTarget - cs.flyStart).Magnitude
            if total > 1 then
                local remaining = (hr.Position - cs.flyTarget).Magnitude
                pct = math.clamp((1 - remaining / total) * 50, 0, 50)
            end
        elseif ph == "Taking" then
            pct = math.max(pct, 58)
        elseif ph == "Home" and cs.homeStart and cs.homeTarget then
            local total = (cs.homeTarget - cs.homeStart).Magnitude
            if total > 1 then
                local remaining = (hr.Position - cs.homeTarget).Magnitude
                pct = math.clamp(58 + (1 - remaining / total) * 40, 58, 98)
            else
                pct = 98
            end
        end
        cs.percent = pct
    end)
end

local function stopProgressMonitor()
    if progressConn then progressConn:Disconnect(); progressConn = nil end
end

local function runInstantFlow(token, egg, e)
    S.currentSteal.phase = "Flying"
    local hr0 = rt()
    if hr0 then S.currentSteal.flyStart = hr0.Position end
    updateStealCard()

    local goOk, goErr = go(e)
    if not goOk then return false, "Fly: " .. tostring(goErr) end
    if isCancel() then return false, "Stopped" end

    S.currentSteal.phase = "Taking"
    S.currentSteal.percent = math.max(S.currentSteal.percent or 0, 55)
    updateStealCard()

    local tkOk, tkErr = tk(e)
    if not tkOk then return false, "Take: " .. tostring(tkErr) end
    if isCancel() then return false, "Stopped" end

    notify("Carrying", "Carrying " .. tostring(egg.Name) .. " home")
    S.currentSteal.phase = "Home"
    updateStealCard()

    local hr1 = rt()
    if hr1 then
        S.currentSteal.homeStart = hr1.Position
        local l = sp()
        if l then
            local lv, lq = l.CFrame.LookVector, l.Position
            local q = hr1.Position - lv * (lv:Dot(hr1.Position - lq) - 48)
            S.currentSteal.homeTarget = Vector3.new(q.X, lq.Y + 3, q.Z)
        end
    end

    local hmOk, hmErr = hm(true)
    if not hmOk then return false, "Home: " .. tostring(hmErr) end

    removeVisualClone()
    S.currentSteal.phase = "Done"
    S.currentSteal.percent = 100
    updateStealCard()
    return true
end

local function runFlow(token)
    local egg = S.selectedEgg
    if not egg then return false, "Select a card first!" end
    local e = { u = egg.Uid, p = egg.Position, r = egg.Raw or {} }
    if not e.p or typeof(e.p) ~= "Vector3" then return false, "Invalid position." end
    -- AreaId check skipped, tk() handles nil safely

    local modeName = (S.stealMode == "admin") and "Admin Abuse TP" or "Instant Steal"
    S.currentSteal = {
        egg = egg, uid = egg.Uid, phase = "Starting", percent = 0,
        flyStart = nil, flyTarget = e.p, homeStart = nil, homeTarget = nil,
    }
    updateStealCard()
    notify("Steal [" .. modeName .. "]", "Starting: " .. tostring(egg.Name))

    S.currentSteal.phase = "Leaving Treadmill"
    updateStealCard()
    askDoffWithRetry(true)
    task.wait(0.3)
    if isCancel() then return false, "Stopped" end

    local okSwap = false
    for _ = 1, 8 do
        if isCancel() then return false, "Stopped" end
        if sw() then okSwap = true; break end
        task.wait(0.2)
    end
    if not okSwap then return false, "Swap failed" end
    if isCancel() then return false, "Stopped" end

    local hr0 = rt()
    lockCameraAt(e.p, hr0 and hr0.Position or nil)
    hideCharacter()
    hideTrailObjects()

    if S.stealMode == "admin" then
        return runAdminAbuseFlow(token, egg, e)
    else
        return runInstantFlow(token, egg, e)
    end
end

-- START / STOP / AUTO
local autoTick

local function startStealFlow()
    if activeFlow then return false end
    if not S.selectedUid or not S.selectedEgg then return false end

    FLOW_TOKEN += 1
    local token = FLOW_TOKEN
    activeFlow = true
    st.on = true
    setRunning(true)
    startProgressMonitor()

    task.spawn(function()
        local ok1, res1, res2 = xpcall(function() return runFlow(token) end, debug.traceback)
        if not ok1 then warn("[ZH] " .. tostring(res1)) end

        local success = ok1 and res1 == true
        local currentEgg = S.currentSteal and S.currentSteal.egg

        if success and currentEgg then
            local okH, errH = pcall(addHistory, currentEgg)
            if not okH then warn("[ZH] addHistory: " .. tostring(errH)) end
        end

        if token ~= FLOW_TOKEN then return end

        if S.currentSteal then
            S.currentSteal.percent = 100
            S.currentSteal.phase = success and "Done" or ("Failed: " .. tostring(ok1 and res2 or "Error"))
            updateStealCard()
        end

        setStatus("Restoring humanoid...")
        restoreVisuals()
        stopProgressMonitor()

        task.wait(0.6)
        S.currentSteal = nil
        updateStealCard()
        st.on = false
        setRunning(false)
        activeFlow = false
        savePersistData()
        task.delay(1, restoreStatus)
        updateSettingsStatus()

        if S.autoEnabled and not cleanedUp then
            local waitTime = success and CONFIG.AUTO_DELAY_SUCCESS or CONFIG.AUTO_DELAY_FAIL
            task.delay(waitTime, function()
                if S.autoEnabled and not cleanedUp and not activeFlow then
                    autoTick()
                end
            end)
        end
    end)
    return true
end

autoTick = function()
    if not S.autoEnabled or cleanedUp or activeFlow then return end
    if not gui.Parent then return end

    task.wait(1)
    if not S.autoEnabled or cleanedUp or activeFlow then return end

    refreshEggs(true)
    renderCards()

    local best = nil
    for _, egg in ipairs(S.filtered) do
        best = egg
        break
    end

    if not best then
        notify("Auto", "No matching card, waiting...", 2)
        task.delay(CONFIG.AUTO_RETRY_NO_EGG, function()
            if S.autoEnabled and not cleanedUp and not activeFlow then
                autoTick()
            end
        end)
        return
    end

    S.selectedUid = best.Uid
    S.selectedEgg = best
    renderCards()

    if not startStealFlow() then
        task.delay(CONFIG.AUTO_RETRY_NO_EGG, function()
            if S.autoEnabled and not cleanedUp and not activeFlow then
                autoTick()
            end
        end)
    end
end

-- PANEL TOGGLE + DRAG
local panelOpen = true
local function setPanelOpen(open)
    panelOpen = open
    if open then
        panel.Visible = true
        panel.Size = UDim2.fromOffset(PANEL_W - 16, PANEL_H - 16)
        -- FIX: Enum.EasingStyle.Back, Enum.EasingDirection.Out
        TweenService:Create(panel, TweenInfo.new(0.2, Enum.EasingStyle.Back, Enum.EasingDirection.Out), { Size = UDim2.fromOffset(PANEL_W, PANEL_H) }):Play()
    else
        local t = TweenService:Create(panel, TweenInfo.new(0.15, Enum.EasingStyle.Quart, Enum.EasingDirection.In), { Size = UDim2.fromOffset(PANEL_W - 16, PANEL_H - 16) })
        t:Play()
        t.Completed:Connect(function()
            if not panelOpen and panel.Parent then panel.Visible = false end
        end)
    end
end

local dragConns = {}

-- Draggable open/close icon (click without moving still toggles the panel)
do
    local iconDragging, iconMoved, iconStart, iconStartPos, iconInput = false, false, nil, nil, nil
    dragConns[#dragConns + 1] = icon.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            iconDragging = true
            iconMoved = false
            iconStart = input.Position
            iconStartPos = icon.Position
            iconInput = input
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    if iconDragging and not iconMoved then
                        setPanelOpen(not panelOpen)
                    end
                    iconDragging = false
                end
            end)
        end
    end)
    dragConns[#dragConns + 1] = UserInputService.InputChanged:Connect(function(input)
        if not iconDragging then return end
        if input.UserInputType ~= Enum.UserInputType.MouseMovement and input ~= iconInput then return end
        local delta = input.Position - iconStart
        if not iconMoved and delta.Magnitude < 6 then return end
        iconMoved = true
        local sc = uiScale.Scale
        icon.Position = UDim2.new(
            iconStartPos.X.Scale, iconStartPos.X.Offset + delta.X / sc,
            iconStartPos.Y.Scale, iconStartPos.Y.Offset + delta.Y / sc
        )
    end)
end
do
    local dragging, dragStart, startPos, dragInput = false
    dragConns[#dragConns + 1] = header.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = panel.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    dragConns[#dragConns + 1] = header.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    dragConns[#dragConns + 1] = UserInputService.InputChanged:Connect(function(input)
        if dragging and dragInput and input == dragInput then
            local delta = input.Position - dragStart
            panel.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)
end

genv.ZincHubCleanup = function()
